// ── Animated Star Field ────────────────────────────────────────
function StarField() {
  const ref = useRef(null);
  useEffect(() => {
    const canvas = ref.current;
    if (!canvas) return;
    const ctx = canvas.getContext("2d");
    let raf;
    const resize = () => { canvas.width = window.innerWidth; canvas.height = window.innerHeight; };
    resize();
    window.addEventListener("resize", resize);

    const stars = Array.from({ length: 260 }, () => ({
      x: Math.random() * window.innerWidth,
      y: Math.random() * window.innerHeight,
      r: Math.random() * 1.4 + 0.2,
      vx: Math.random() * 0.38 + 0.06,
      vy: (Math.random() - 0.5) * 0.07,
      depth: Math.random() * 2.4 + 0.4,
      phase: Math.random() * Math.PI * 2,
      spd: Math.random() * 0.028 + 0.014,
      col: Math.random() > 0.88 ? `hsl(${185 + Math.floor(Math.random()*60)},80%,88%)` : "#fff",
    }));

    const shooters = [];
    let nextShot = 160 + Math.random() * 180;

    const mkShooter = () => ({
      x: -60,
      y: Math.random() * canvas.height * 0.55,
      vx: 11 + Math.random() * 9,
      vy: 2 + Math.random() * 3.5,
      len: 100 + Math.random() * 130,
      alpha: 1,
      decay: 0.016 + Math.random() * 0.018,
    });

    const draw = () => {
      ctx.clearRect(0, 0, canvas.width, canvas.height);
      for (const s of stars) {
        s.phase += s.spd;
        const a = Math.max(0, Math.min(1, 0.28 + Math.sin(s.phase) * 0.46));
        s.x += s.vx * (s.depth / 2.5);
        s.y += s.vy;
        if (s.x > canvas.width  + 4) s.x = -4;
        if (s.y > canvas.height + 4) s.y = -4;
        if (s.y < -4) s.y = canvas.height + 4;
        ctx.globalAlpha = a;
        ctx.fillStyle = s.col;
        ctx.beginPath();
        ctx.arc(s.x, s.y, s.r * (s.depth / 2.5), 0, Math.PI * 2);
        ctx.fill();
        if (s.r > 1.1 && s.depth > 2 && a > 0.62) {
          ctx.globalAlpha = a * 0.09;
          ctx.beginPath();
          ctx.arc(s.x, s.y, s.r * 5, 0, Math.PI * 2);
          ctx.fill();
        }
      }
      ctx.globalAlpha = 1;
      nextShot--;
      if (nextShot <= 0) { shooters.push(mkShooter()); nextShot = 200 + Math.random() * 340; }
      for (let i = shooters.length - 1; i >= 0; i--) {
        const sh = shooters[i];
        const x0 = sh.x - sh.vx * (sh.len / 14);
        const y0 = sh.y - sh.vy * (sh.len / 14);
        const g = ctx.createLinearGradient(x0, y0, sh.x, sh.y);
        g.addColorStop(0, "rgba(0,229,255,0)");
        g.addColorStop(0.7, `rgba(180,240,255,${sh.alpha * 0.55})`);
        g.addColorStop(1, `rgba(255,255,255,${sh.alpha})`);
        ctx.strokeStyle = g;
        ctx.lineWidth = 1.6;
        ctx.globalAlpha = sh.alpha;
        ctx.beginPath();
        ctx.moveTo(x0, y0);
        ctx.lineTo(sh.x, sh.y);
        ctx.stroke();
        ctx.globalAlpha = 1;
        sh.x += sh.vx; sh.y += sh.vy; sh.alpha -= sh.decay;
        if (sh.alpha <= 0 || sh.x > canvas.width + 200) shooters.splice(i, 1);
      }
      raf = requestAnimationFrame(draw);
    };
    draw();
    return () => { cancelAnimationFrame(raf); window.removeEventListener("resize", resize); };
  }, []);
  return ;
}
